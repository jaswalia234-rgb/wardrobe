function renderWardrobe() { const container = document.getElementById("wardrobe-items"); container.innerHTML = ""; if (!closetData || closetData.length === 0) { container.innerHTML = `

No items found in database. Check that your Notion database has rows added!

`; return; } const categories = { Outerwear: [], Top: [], Bottom: [], Shoes: [], Other: [] }; closetData.forEach(item => { let cat = item.properties.Category?.select?.name || "Other"; // Auto-fix plural & variant tags if (cat === "Tops" || cat === "Shirt" || cat === "Shirts") cat = "Top"; if (cat === "Bottoms" || cat === "Pants" || cat === "Trousers" || cat === "Jeans" || cat === "Skirts") cat = "Bottom"; if (cat === "Jackets" || cat === "Coats" || cat === "Cardigans") cat = "Outerwear"; if (cat === "Footwear" || cat === "Boots" || cat === "Sneakers") cat = "Shoes"; if (categories[cat]) { categories[cat].push(item); } else { categories.Other.push(item); } }); for (const [category, items] of Object.entries(categories)) { if (items.length === 0) continue; const section = document.createElement("div"); section.className = "category-section"; section.innerHTML = `

${category}${category === 'Other' ? '' : 's'} (${items.length})

`; container.appendChild(section); const carousel = section.querySelector(`.carousel`); items.forEach(clothing => { const name = clothing.properties.Name?.title[0]?.plain_text || "Untitled"; const imgUrl = clothing.properties.Photo?.files[0]?.file?.url || clothing.properties.Photo?.files[0]?.external?.url || ""; const lastWashed = clothing.properties["Last Washed"]?.date?.start; const wears = clothing.properties["Wears Since Wash"]?.number || 0; const days = calculateDays(lastWashed); let tagClass = "tag-fresh"; let tagText = `${wears} wears · Washed ${days === 999 ? 'N/A' : days + 'd ago'}`; if (wears >= 3 || days > 7) tagClass = "tag-warn"; if (wears >= 5 || days > 14) { tagClass = "tag-wash"; tagText = "Wash Recommended!"; } const isEquipped = activeOutfit[category] === clothing.id; const card = document.createElement("div"); card.className = `card ${isEquipped ? 'equipped' : ''}`; card.onclick = () => tryOn(clothing.id, category); card.innerHTML = ` ${imgUrl ? `` : `
No Image
`}
${name}
${tagText}
`; carousel.appendChild(card); }); } }
